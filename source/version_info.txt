# Windows version resource, compiled into the exe by PyInstaller
# (DesktopTasks.spec passes it as version=).
#
# Without this the binary carries an icon but no name. Windows reads
# FileDescription wherever it has to name a program it did not launch
# itself: Task Manager's Startup tab, Settings > Apps > Startup, the
# Details tab, and the UAC-style prompts. An exe with no version
# resource shows up there as a blank name with no publisher, which is
# exactly what a user should not see next to something that starts with
# their computer.
#
# The icon in those lists comes from the exe too, which is why a Startup
# entry for a source checkout stays generic: its shortcut points at
# pythonw.exe, and Windows describes the target, not the .lnk.
#
# filevers and prodvers are (major, minor, patch, build) and must be
# four integers. Keep them in step with the strings below.
VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(0, 1, 0, 0),
    prodvers=(0, 1, 0, 0),
    # Which of the fields below are valid, and the file's type. 0x3f is
    # all of them; 0x1 is VFT_APP, an application.
    mask=0x3f,
    flags=0x0,
    # VOS_NT_WINDOWS32.
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0),
  ),
  kids=[
    StringFileInfo([
      StringTable(
        # Language and charset: 0409 is US English, 04b0 is Unicode.
        '040904b0',
        [
          # The one Windows actually shows in a program list.
          StringStruct('FileDescription', 'DesktopTasks'),
          StringStruct('ProductName', 'DesktopTasks'),
          StringStruct('CompanyName', 'DesktopTasks'),
          StringStruct('FileVersion', '0.1.0.0'),
          StringStruct('ProductVersion', '0.1.0.0'),
          StringStruct('InternalName', 'DesktopTasks'),
          StringStruct('OriginalFilename', 'DesktopTasks.exe'),
          StringStruct('LegalCopyright', ''),
          StringStruct('Comments', 'Your Google Tasks, pinned to the desktop.'),
        ])
    ]),
    VarFileInfo([VarStruct('Translation', [0x409, 1200])])
  ]
)
